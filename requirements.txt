# Python 3.12 or newer is required. Try to keep within the 3.12.x series. 
# bs-roformer-infer will not install on anything older, 
# and some of its dependencies use `X | None` type hints that
# only work at runtime from 3.12.0 and up. 3.12.10 is recommended.
#
# Install into a virtual environment:
#     py -3.12 -m venv venv
#     venv\Scripts\activate            (Windows)
#     source venv/bin/activate         (Linux / macOS)
#     python -m pip install --upgrade pip
#     python -m pip install -r requirements.txt
#
# PyTorch is listed here as the plain CPU package. For GPU separation, install
# the CUDA build FIRST, using the command from https://pytorch.org for your
# driver's CUDA version, e.g.:
#     python -m pip install torch --index-url https://download.pytorch.org/whl/cu126
# then run the line above; pip will keep the CUDA build already installed.
# Check with:
#     python -c "import torch; print(torch.__version__, torch.cuda.is_available())"
# A version ending in +cu126 and True means the GPU path is active.
# ---------------------------------------------------------------------------

pydub>=0.25

# --- Mel-Band models (karaoke) ---------------------------------------------
# Required for any Mel-Band RoFormer, such as becruily's karaoke model: those
# models ask librosa.filters.mel how many FFT bins belong to each band, and
# those counts set the width of every mask-estimator layer. RAMMA's built-in
# filterbank is close but not identical, so the weights will not load without
# this. Not needed if you only use BS-RoFormer models.
librosa>=0.10
